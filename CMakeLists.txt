cmake_minimum_required(VERSION 3.20)
project(best_ticket_kiosk C)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror)

include_directories(include generated)
set(CORE_SOURCES src/catalog.c src/booking.c src/i18n.c src/mapview.c src/search.c)

find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(SDL2_TTF IMPORTED_TARGET SDL2_ttf)
find_package(ZLIB REQUIRED)
if(SDL2_TTF_FOUND)
  find_package(PNG REQUIRED)
  add_executable(best-kiosk src/main.c src/ui_sdl.c src/ticket_qr.c src/ticket_pdf.c ${CORE_SOURCES})
  target_link_libraries(best-kiosk PRIVATE PkgConfig::SDL2 PkgConfig::SDL2_TTF PNG::PNG ZLIB::ZLIB)
  if(WIN32)
    target_link_libraries(best-kiosk PRIVATE shell32)
  endif()
  if(NOT WIN32)
    target_link_libraries(best-kiosk PRIVATE m)
  endif()
else()
  message(WARNING "SDL2_ttf is missing: core tests remain available; install it to build the graphical kiosk.")
endif()

enable_testing()
foreach(test_name booking catalog i18n mapview search)
  add_executable(test-${test_name} tests/test_${test_name}.c ${CORE_SOURCES})
  add_test(NAME ${test_name} COMMAND test-${test_name})
endforeach()

add_executable(test-ticketing tests/test_ticketing.c ${CORE_SOURCES} src/ticket_qr.c)
add_test(NAME ticketing COMMAND test-ticketing)

add_executable(test-qr tests/test_qr.c src/ticket_qr.c)
add_test(NAME qr COMMAND test-qr)
